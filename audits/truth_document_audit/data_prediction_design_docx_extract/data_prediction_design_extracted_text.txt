The 50/50 Split Rule: For Limited Entry permits, 50% of tags go to the highest point holders (Bonus Draw), while the remaining 50% are drawn randomly among all applicants (Random Draw).
Monte Carlo Simulation Engine
The simulator must run thousands of iterations per unit to account for the "Random Draw" portion of Utah's system.
import numpy as np
def simulate_le_draw(applicants, permits, iterations=10000):
"""
applicants: List of (points, applicant_id)
permits: Total available for the unit
"""
results = []
bonus_permits = permits // 2
random_permits = permits - bonus_permits
for _ in range(iterations):
# 1. Bonus Draw (Highest points win)
applicants.sort(key=lambda x: x[0], reverse=True)
winners = [a[1] for a in applicants[:bonus_permits]]
remaining = applicants[bonus_permits:]
# 2. Random Draw (Bonus points squared + 1)
# Utah uses (Points + 1) chances.
# Note: Some states square them, verify UDWR current rule.
weights = [a[0] + 1 for a in remaining]
weights = np.array(weights) / sum(weights)
random_winners = np.random.choice(
[a[1] for a in remaining],
size=min(random_permits, len(remaining)),
replace=False,
p=weights
)
winners.extend(random_winners)
results.append(winners)
return calculate_odds(results)
